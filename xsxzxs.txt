import SubTabs from "./SubTabs";
import styles from "./Header.module.css";
import { NavLink } from "react-router-dom";
import { useState } from "react";
import { useEffect } from "react";

const Header = ({ isHome }) => {

  const [showHeader, setShowHeader] = useState(true);
  const [solid, setSolid] = useState(false);

  // Home 전용 스크롤 이벤트
  useEffect(() => {
    if (!isHome) {
      setShowHeader(true);
      setSolid(true);
      return;
    }

    const threshold = 10;
    const holdShow = 120;
    let prev = window.scrollY;

    const handleScroll = () => {
      const cur = window.scrollY;
      const diff = cur - prev;

      if (Math.abs(diff) < threshold) return;

      if (diff < 0) setShowHeader(true);
      if (diff > 0 && cur > holdShow) setShowHeader(false);

      setSolid(cur > 150);

      prev = cur;
    };

    window.addEventListener("scroll", handleScroll);
    return () => window.removeEventListener("scroll", handleScroll);

  }, [isHome]);

  // 공용 className 활성화 처리 로직
  const cx = (base, active) => ({ isActive }) =>
    isActive ? `${base} ${active}` : base;

  return (
    <header
      className={`
    ${styles.header}
    ${isHome ? (solid ? styles.solid : styles.transparent) : styles.subHeader}
    ${isHome ? styles.homeText : ""}
    ${showHeader ? "" : styles.hide}
  `}
    >
      <div className={styles.logoArea}>
        <a href="/" className={styles.logo}>TNT</a>
      </div>

      <div className={styles.tabArea}>

        {/* 메인탭 */}
        <nav className={styles.mainTabs}>
          <NavLink to="/color" className={cx(styles.mainTab, styles.mainTabActive)}>Personal Color</NavLink>
          <NavLink to="/body" className={cx(styles.mainTab, styles.mainTabActive)}>Personal Body</NavLink>
          <NavLink to="/fitroom" className={cx(styles.mainTab, styles.mainTabActive)}>Fitting Room</NavLink>
          {/* 여기에 커뮤니티 경로 넣으세요 
          <NavLink to="/community"  className={cx(styles.mainTab, styles.mainTabActive)}>Community</NavLink>*/}
        </nav>

        {/* 서브탭 */}
        {!isHome && (
          <div className={styles.subTabsWrapper}>
            <SubTabs cx={cx} />
          </div>
        )}
      </div>
    </header>
  );
};

export default Header;
